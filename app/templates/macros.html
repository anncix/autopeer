{# Reusable presentation macros shared by every page. - badge / status_badge / deploy_badge / bool_pill: coloured pills for states - stat_card: dashboard metric card - page_header: title + optional subtitle - code_block: 
 with a copy button (multi-line configs)
  - copy_button: standalone copy button (short values like tokens/keys)
  - pagination: prev / page-of / next pager
#}

{% macro badge(text, tone="neutral", i18n_key=None) %}{{ text }}{% endmacro %}

{% macro status_badge(status) %}
  {%- set tones = {"approved": "ok", "disabled": "bad", "pending": "warn"} -%}
  {{ badge(status, tones.get(status, "neutral"), "status." ~ status) }}
{%- endmacro %}

{% macro deploy_badge(status) %}
  {%- set tones = {"deployed": "ok", "failed": "bad", "deploying": "warn"} -%}
  {{ badge(status.replace("_", " "), tones.get(status, "neutral"), "status." ~ status) }}
{%- endmacro %}

{% macro bool_pill(value, on="yes", off="no", i18n_prefix=None) %}
  {{ badge(on, "ok", (i18n_prefix ~ on) if i18n_prefix else none) if value else badge(off, "neutral", (i18n_prefix ~ off) if i18n_prefix else none) }}
{%- endmacro %}

{% macro stat_card(label, value, sub=None, tone="neutral", href=None, i18n_key=None, i18n_sub_key=None, i18n_label_params=None, i18n_sub_params=None) %}

{% endif %}
    
{{ value }}

    
{{ label }}

    {% if sub %}
{{ sub }}
{% endif %}
  {% if href %}{% else %}
{% endif %}
{%- endmacro %}
